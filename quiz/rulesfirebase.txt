rules_version = '2';

// ═══════════════════════════════════════════════════════════════════
// Ovelhas Coloridas — Regras de Segurança do Cloud Firestore
// Projeto: ovelhascoloridas-235c1
// ═══════════════════════════════════════════════════════════════════
//
// ARQUITETURA DE SEGURANÇA
// O site opera sem sistema de autenticação — os painéis de admin
// (admin.html do site e admin.html do Quiz Studio) são acedidos
// diretamente por URL, sem login. Tanto o site público como os
// painéis correm, por isso, com permissões de cliente. As regras
// abaixo validam o formato e o tamanho de todos os dados escritos,
// de forma a limitar abusos sem bloquear funcionalidade legítima.
//
// COLEÇÕES
//   Site:  reviews · review_tokens
//   Quiz:  quiz (estado do jogo em direto)
//          quiz_kits (biblioteca de quizzes)
//          quiz_media (metadados + miniaturas das imagens)
//          quiz_media_data (imagens completas, em base64)
//
// NOTA: Para uma segurança mais robusta, considerar no futuro
// adicionar Firebase Authentication aos painéis de administração.
// Tudo o que não estiver explicitamente permitido está bloqueado.
// ═══════════════════════════════════════════════════════════════════

service cloud.firestore {
  match /databases/{database}/documents {

    // ═════════════════════ SITE: LIVRO ═════════════════════

    // ─── Coleção: reviews ─────────────────────────────────
    // Armazena críticas e testemunhos submetidos pelos leitores.
    match /reviews/{reviewId} {

      // Qualquer visitante pode LER críticas aprovadas (visíveis no site)
      allow read: if resource.data.approved == true;

      // Qualquer visitante pode CRIAR uma nova crítica,
      // desde que os dados sejam válidos (validação anti-spam).
      allow create: if validateReview(request.resource.data);

      // UPDATE e DELETE: necessários para o painel de administração
      // (aprovar/ocultar críticas, alterar badge de Compra Confirmada).
      // AVISO: sem auth, estas operações são abertas do lado do cliente.
      allow update: if validateReviewUpdate(request.resource.data);
      allow delete: if true;
    }

    // ─── Coleção: review_tokens ───────────────────────────
    // Tokens de "Compra Confirmada" gerados pelo admin.
    match /review_tokens/{tokenId} {

      // Leitura pública — necessária para que a página avaliar.html
      // possa validar o token antes da submissão.
      allow read: if true;

      // Criação de tokens: painel de admin (sem auth, é aberto).
      allow create: if validateToken(request.resource.data);

      // Update: necessário para incrementar useCount quando um token é usado.
      allow update: if true;

      // Delete: para o admin apagar tokens inválidos/expirados.
      allow delete: if true;
    }


    // ═════════════════════ QUIZ STUDIO ═════════════════════

    // ─── Coleção: quiz ────────────────────────────────────
    // Documento único "main" com o estado do jogo em direto.
    // Lido pelo display.html (projetor) e escrito pelo admin.html.
    match /quiz/{docId} {
      allow read: if true;

      // Só o documento "main" existe; o estado é validado a cada escrita.
      allow create, update: if docId == 'main'
                            && validateLive(request.resource.data);

      // O estado do jogo nunca deve ser apagado.
      allow delete: if false;
    }

    // ─── Coleção: quiz_kits ───────────────────────────────
    // Biblioteca de quizzes (título + lista de perguntas).
    match /quiz_kits/{kitId} {
      allow read: if true;
      allow create, update: if validateKit(request.resource.data);
      allow delete: if true;
    }

    // ─── Coleção: quiz_media ──────────────────────────────
    // Metadados e miniatura (leve) de cada imagem da biblioteca.
    match /quiz_media/{mediaId} {
      allow read: if true;
      allow create: if validateMediaMeta(request.resource.data);

      // Só é permitido renomear a imagem.
      allow update: if validateMediaMeta(request.resource.data);
      allow delete: if true;
    }

    // ─── Coleção: quiz_media_data ─────────────────────────
    // Imagem completa (base64, já otimizada pelo admin). Limite de
    // ~900 KB por documento (limite do Firestore: 1 MiB).
    match /quiz_media_data/{mediaId} {
      allow read: if true;
      allow create: if validateMediaData(request.resource.data);

      // As imagens são imutáveis: para alterar, carrega-se uma nova.
      allow update: if false;
      allow delete: if true;
    }


    // ═════════════════════ FUNÇÕES: SITE ═════════════════════

    // ─── Validar dados de uma nova crítica ────────────────
    // Garante campos obrigatórios com tipos e tamanhos corretos.
    function validateReview(data) {
      return (
        // Campos obrigatórios presentes
        data.keys().hasAll(['name', 'rating', 'comment', 'character',
                            'verifiedPurchase', 'approved'])

        // name: string, 1 a 60 caracteres
        && data.name is string
        && data.name.size() >= 1
        && data.name.size() <= 60

        // rating: número de 0 a 5 (estrelas)
        && data.rating is number
        && data.rating >= 0
        && data.rating <= 5

        // comment: string obrigatório, 1 a 1200 caracteres
        && data.comment is string
        && data.comment.size() >= 1
        && data.comment.size() <= 1200

        // title: opcional mas se presente deve ser string <= 100 chars
        && (!data.keys().hasAny(['title']) || (data.title is string && data.title.size() <= 100))

        // character: deve ser uma das personagens válidas
        && data.character is string
        && data.character in [
            'ovelha-arcoiris', 'ovelha-azul', 'ovelha-roxa',
            'ovelha-melancia', 'ovelha-verde', 'ovelha-rosa',
            'ovelha-laranja', 'elefante-amendoim'
          ]

        // verifiedPurchase: deve ser boolean
        && data.verifiedPurchase is bool

        // approved: novas críticas entram sempre como aprovadas (true)
        && data.approved == true
      );
    }

    // ─── Validar update de uma crítica (pelo admin) ───────
    function validateReviewUpdate(data) {
      return (
        // Para updates do admin, apenas verificamos campos mínimos
        data.keys().hasAll(['approved'])
        && data.approved is bool
      );
    }

    // ─── Validar criação de um token ──────────────────────
    function validateToken(data) {
      return (
        data.keys().hasAll(['token', 'active', 'useCount', 'maxUses'])
        && data.token is string
        && data.token.size() >= 10
        && data.token.size() <= 30
        && data.active is bool
        && data.useCount is number
        && data.useCount >= 0
        && data.maxUses is number
        && data.maxUses >= 1
      );
    }


    // ═════════════════════ FUNÇÕES: QUIZ ═════════════════════

    // ─── Validar uma pergunta ─────────────────────────────
    // Tipos: quiz4 · quiz3 · quiz2 · tf (Verdadeiro/Falso)
    function validateQuestion(q) {
      return (
        q is map
        && q.keys().hasAll(['type', 'text', 'answers', 'correct', 'time', 'pointsMode', 'points'])

        && q.type in ['quiz4', 'quiz3', 'quiz2', 'tf']

        // Texto da pergunta (pode estar vazio durante a edição)
        && q.text is string
        && q.text.size() <= 300

        // Entre 2 e 4 respostas, de texto curto
        && q.answers is list
        && q.answers.size() >= 2
        && q.answers.size() <= 4

        // Índice da resposta correta dentro do intervalo
        && q.correct is int
        && q.correct >= 0
        && q.correct < q.answers.size()

        // Tempo em segundos (0 = sem cronómetro)
        && q.time is number
        && q.time >= 0
        && q.time <= 600

        // Pontuação
        && q.pointsMode in ['standard', 'double', 'none']
        && q.points is number
        && q.points >= 0
        && q.points <= 100000

        // Campos opcionais
        && (!q.keys().hasAny(['image'])       || (q.image is string && q.image.size() <= 600))
        && (!q.keys().hasAny(['explanation']) || (q.explanation is string && q.explanation.size() <= 500))
      );
    }

    // ─── Validar estado do jogo em direto ─────────────────
    // As escritas parciais (update) são avaliadas sobre o documento
    // final resultante, pelo que basta validar o resultado.
    function validateLive(data) {
      return (
        data.keys().hasAll(['phase', 'qIndex', 'teams', 'scores', 'questions'])

        && data.phase in ['lobby', 'question', 'reveal', 'board', 'final']

        && data.qIndex is int
        && data.qIndex >= 0
        && data.qIndex < 500

        // Grupos (máx. 30)
        && data.teams is list
        && data.teams.size() <= 30

        // Pontos e respostas por grupo
        && data.scores is map
        && (!data.keys().hasAny(['picks'])    || data.picks is map)
        && (!data.keys().hasAny(['awarded'])  || data.awarded is map)

        // Cópia das perguntas do quiz em jogo (máx. 200)
        && data.questions is list
        && data.questions.size() <= 200
        && (!data.keys().hasAny(['kitTitle']) || (data.kitTitle is string && data.kitTitle.size() <= 120))
      );
    }

    // ─── Validar um quiz da biblioteca ────────────────────
    function validateKit(data) {
      return (
        data.keys().hasAll(['title', 'questions'])
        && data.title is string
        && data.title.size() >= 1
        && data.title.size() <= 120
        && data.questions is list
        && data.questions.size() <= 200
        && (data.questions.size() == 0 || validateQuestion(data.questions[0]))
      );
    }

    // ─── Validar metadados de uma imagem ──────────────────
    function validateMediaMeta(data) {
      return (
        data.keys().hasAll(['name', 'thumb'])
        && data.name is string
        && data.name.size() >= 1
        && data.name.size() <= 120
        && data.thumb is string
        && data.thumb[0:11] == 'data:image/'
        && data.thumb.size() <= 150000
      );
    }

    // ─── Validar imagem completa (base64) ─────────────────
    function validateMediaData(data) {
      return (
        data.keys().hasOnly(['data'])
        && data.data is string
        && data.data[0:11] == 'data:image/'
        && data.data.size() <= 900000
      );
    }

  }
}
